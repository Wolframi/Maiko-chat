Нативное изменение размеров окна Electron — Майко
Основное окно: frame: false, thickFrame: true, resizable: true, transparent: false, roundedCorners: false.
Windows управляет изменением размеров, разворачиванием и восстановлением. Ручные обработчики мыши, IPC ресайза и трекер удалены.
События will-resize и resize подгоняют встроенную страницу под размер окна.
Полные исходные файлы. Некоторые файлы также содержат код других функций приложения.

const uiUrl = pathToFileURL(path.join(__dirname, 'ui', 'index.html')).href;
const siteLayoutCSS = fs.readFileSync(path.join(__dirname, 'ui/site-layout.css'), 'utf8') + '\n' +

const compactMembersJS = fs.readFileSync(path.join(__dirname, 'ui/compact-members.js'), 'utf8');
const downloadProgressJS = fs.readFileSync(path.join(__dirname, 'ui/download-progress.js'), 'utf8');
const settingsPath = () => path.join(app.getPath('userData'), 'desktop-settings.json');
function log(message) {
  try { fs.appendFileSync(path.join(app.getPath('userData'), 'desktop.log'), `${new Date().toISOString()} ${message}\n`); } catch { /* logging is optional */ }
}
function publish(patch = {}) {
  const next = { ...state, ...patch, maximized: !!win && !win.isDestroyed() && (win.isMaximized() || win.isFullScreen()) };
  if (Object.keys(next).every((key) => next[key] === state[key])) return;
  state = next;
  if (win && !win.isDestroyed()) win.webContents.send('desktop:state-changed', state);
  downloadsPanel?.publish(state.downloads);
  if (patch.downloads && site && !site.webContents.isDestroyed() && trusted(site.webContents.getURL())) {
    site.webContents.send('desktop:site-downloads-changed', downloads?.siteProgress() || []);
  }
}
function layout() {
  if (!win || win.isDestroyed()) return;
  const expanded = win.isMaximized() || win.isFullScreen();
  // Keep the native content rectangular in normal and expanded modes.
  if (layoutExpanded !== expanded) {
    win.contentView.setBorderRadius(expanded ? 0 : RADIUS);
    layoutExpanded = expanded;
  }
  if (site) {
    const [width, height] = win.getContentSize();
    sizeSite(width, height, expanded);
  }
  publish();
}
function sizeSite(width, height, expanded) {
  if (!site) return;
  const next = siteBounds(width, height, expanded);
  const current = site.getBounds();
  if (Object.keys(next).some((key) => next[key] !== current[key])) site.setBounds(next);
}
function toggleMaximize() {
  if (win.isFullScreen()) { win.setFullScreen(false); return; }
  if (win.isMaximized()) win.unmaximize();
  else win.maximize();
}
function trusted(url) { return state.url && isSameOrigin(url, state.url); }
function requireShell(event) {
  if (!win || event.sender !== win.webContents || event.senderFrame !== win.webContents.mainFrame || event.senderFrame.url !== uiUrl) {
    throw new Error('Недоступно для этой страницы');
  }
}
function requireSite(event) {
  if (!site || event.sender !== site.webContents || event.senderFrame !== site.webContents.mainFrame || !trusted(event.senderFrame.url)) {
    throw new Error('Недоступно для этой страницы');
  }
}
async function openExternal(url) {
  try {
    const parsed = new URL(url);
    if (['https:', 'http:', 'mailto:'].includes(parsed.protocol)) await shell.openExternal(parsed.href);
  } catch (error) { log(error.message); }
}
function createSite() {
  const view = new WebContentsView({ webPreferences: {
    partition: 'persist:maiko-site', sandbox: true, contextIsolation: true,
    preload: path.join(__dirname, 'site-preload.cjs'),
    nodeIntegration: false, webSecurity: true, backgroundThrottling: false,
  } });
  view.setBackgroundColor('#0c0c0e');
  view.setVisible(false);
  win.contentView.addChildView(view);
  const contents = view.webContents;
  contents.setWindowOpenHandler(({ url }) => {
    if (trusted(url) && new URL(url).pathname.startsWith('/uploads/') && new URL(url).searchParams.has('download')) contents.downloadURL(url);
    else void openExternal(url);
    return { action: 'deny' };
  });
  const guardNavigation = (event, url) => {
    if (!trusted(url)) { event.preventDefault(); void openExternal(url); }
  };
  contents.on('will-navigate', guardNavigation);
  contents.on('will-redirect', guardNavigation);
  contents.on('will-attach-webview', (event) => event.preventDefault());
  contents.on('did-start-navigation', (_event, _url, inPlace, isMainFrame) => {
    if (isMainFrame && !inPlace) publish({ phase: 'loading', message: 'Подключаемся…' });
  });
  const showSite = () => {
    if (!trusted(contents.getURL())) return;
    publish({ phase: 'ready', message: state.mode === 'local' ? 'Локальный сервер' : 'Подключено' });
    view.setVisible(true);
    contents.focus();
  };
  contents.on('dom-ready', () => {
    if (!trusted(contents.getURL())) return;
    void Promise.all([
      contents.insertCSS(siteLayoutCSS),
      contents.executeJavaScriptInIsolatedWorld(1001, [{ code: compactMembersJS }, { code: downloadProgressJS }]),
    ]).then(showSite).catch((error) => {
      log(`Desktop layout helpers unavailable: ${error.message}`);
      showSite();
    });
  });
  contents.on('did-fail-load', (_event, code, description, _url, isMainFrame) => {
    if (!isMainFrame || code === -3) return;
    view.setVisible(false);
    publish({ phase: 'error', message: `Не удалось открыть Майко (${description}). Проверьте соединение и попробуйте снова.` });
  });
  contents.on('render-process-gone', (_event, details) => {
    view.setVisible(false);
    publish({ phase: 'error', message: 'Страница закрылась. Нажмите «Повторить».', detail: details.reason });
  });
  contents.on('before-input-event', (event, input) => {
    if (input.type !== 'keyDown') return;
    if (input.key === 'F5' || ((input.control || input.meta) && input.key.toLowerCase() === 'r')) {
      event.preventDefault(); void connect(state.url);
    }
    if (input.key === 'F11') { event.preventDefault(); win.setFullScreen(!win.isFullScreen()); }
    if (!app.isPackaged && input.key === 'F12') { event.preventDefault(); contents.toggleDevTools(); }
  });
  return view;
}
async function connect(url) {
  const generation = ++loadGeneration;
  site?.setVisible(false);
  publish({ phase: 'loading', message: 'Запускаем Майко…' });
  try {
    log(`Connecting in ${state.mode} mode`);
    let target = url;
    if (state.mode === 'local') {
      if (!localServer) {
        const root = findProjectRoot(process.env.MAIKO_PROJECT_DIR || (app.isPackaged ? path.dirname(process.execPath) : __dirname));
        const port = Number(process.env.MAIKO_LOCAL_PORT || defaults.localPort || 3000);
        if (!Number.isInteger(port) || port < 1024 || port > 65535) throw new Error('localPort должен быть числом от 1024 до 65535.');
        localServer = await startLocalServer({ root, port, utilityProcess, onLog: log, dev: process.argv.includes('--dev') });
      }
      target = localServer.url;
    }
    target = normalizeUrl(target);
    if (generation !== loadGeneration || !win || win.isDestroyed()) return;
    publish({ url: target });
    if (!site) { site = createSite(); layout(); }
    await site.webContents.loadURL(target);
  } catch (error) {
    if (generation !== loadGeneration) return;
    site?.setVisible(false);
    log(error.stack || error.message);
    publish({ phase: 'error', message: error.message });
  }
}
function saveWindow() {
  if (!win || win.isDestroyed()) return;
  fs.writeFileSync(settingsPath(), JSON.stringify({ bounds: win.getNormalBounds(), maximized: win.isMaximized() }, null, 2));
}
function createWindow() {
  const saved = readJson(settingsPath());
  win = new BrowserWindow({
    ...visibleBounds(saved.bounds, screen.getAllDisplays()),
    minWidth: 800, minHeight: 600, frame: false, roundedCorners: false,
    // Windows owns edge hit testing and the native sizing loop.
    transparent: false, thickFrame: true, resizable: true, backgroundColor: '#0c0c0e', title: 'Майко',
    // PNG works reliably through NativeImage, including inside app.asar.
    // The builder uses the multi-size ICO for the executable and shortcuts.
    icon: path.join(__dirname, 'assets', 'icon.png'), show: false, autoHideMenuBar: true,
    webPreferences: { preload: path.join(__dirname, 'preload.cjs'), sandbox: true, contextIsolation: true, nodeIntegration: false, backgroundThrottling: false },
  });
  win.setMenu(null);
  // Cover newly exposed pixels while Chromium draws its next frame.
  // The shell and site's native surfaces use the same dark background.
  win.contentView.setBackgroundColor('#0c0c0e');
  // The shell's native view needs its own dark fallback while its renderer
  // catches up during resizing.
  const shellView = win.contentView.children.find((view) => view.webContents === win.webContents);
  shellView?.setBackgroundColor('#0c0c0e');
  layoutExpanded = undefined;
  win.webContents.setWindowOpenHandler(() => ({ action: 'deny' }));
  win.webContents.on('will-navigate', (event, url) => { if (url !== uiUrl) event.preventDefault(); });
  win.on('will-resize', (_event, bounds) => {
    // Lay out the embedded site before Windows presents a larger client area.
    // The system still owns the window's bounds and the entire sizing gesture.
    const [width, height] = win.getSize();
    const [contentWidth, contentHeight] = win.getContentSize();
    sizeSite(bounds.width - (width - contentWidth), bounds.height - (height - contentHeight), win.isMaximized() || win.isFullScreen());
  });
  win.on('resize', layout);
  win.on('resize', () => downloadsPanel?.hide());
  win.on('blur', () => downloadsPanel?.hide());
  for (const event of ['maximize', 'unmaximize', 'enter-full-screen', 'leave-full-screen']) win.on(event, layout);
  win.on('close', () => { saveWindow(); });
  win.on('closed', () => {
    loadGeneration++;
    downloadsPanel?.dispose();
    if (site && !site.webContents.isDestroyed()) site.webContents.close();
    site = undefined; win = undefined;
  });
  win.once('ready-to-show', () => {
    if (saved.maximized) toggleMaximize();
    layout();
    if (!testing) win.show();
  });
  const createdWindow = win;
  makeWindowRectangular(createdWindow);
  void createdWindow.loadURL(uiUrl).then(() => {
    if (createdWindow.isDestroyed()) return;
    publish();
    return connect((testing || !app.isPackaged ? process.env.MAIKO_DESKTOP_URL : undefined) || defaults.url);
  }).catch((error) => { if (win && !win.isDestroyed()) log(error.stack || error.message); });
}

if (!app.requestSingleInstanceLock()) { app.quit(); }
else {
  app.on('second-instance', () => {
    if (!win) return;
    if (win.isMinimized()) win.restore();
    win.show(); win.focus();
  });
  app.whenReady().then(() => {
    app.setAppUserModelId('io.maiko.desktop');
    Menu.setApplicationMenu(null);
    const siteSession = session.fromPartition('persist:maiko-site');
    configureLocalAssets({ siteSession, trusted, log });
    const allowed = new Set(['media', 'notifications', 'display-capture', 'fullscreen', 'clipboard-sanitized-write']);
    siteSession.setPermissionCheckHandler((contents, permission, origin) =>
      !!site && contents === site.webContents && trusted(origin) && allowed.has(permission));
    siteSession.setPermissionRequestHandler((contents, permission, callback, details) => {
      callback(!!site && contents === site.webContents && trusted(details.requestingUrl || contents.getURL()) && allowed.has(permission));
    });
    configureCapture({ siteSession, getWindow: () => win, getSite: () => site, trusted, testing, log });
    downloads = createDownloads({
      directory: path.join(testing && process.env.MAIKO_DESKTOP_DOWNLOADS_DIR || app.getPath('downloads'), 'Maiko'),
      historyPath: path.join(app.getPath('userData'), 'downloads.json'),
      siteSession, getSite: () => site, trusted, publish, shell, log,
    });
    downloadsPanel = configureDownloadsPanel({ ipcMain, getWindow: () => win, requireShell, downloads, log });
    ipcMain.handle('desktop:site-downloads-state', (event) => { requireSite(event); return downloads.siteProgress(); });
    ipcMain.handle('desktop:site-downloads-cancel', (event, id) => { requireSite(event); return downloads.action('cancel', id); });
    ipcMain.handle('desktop:downloads', (event, command, id) => {
      requireShell(event);
      return downloads.action(command, id);
    });
    updates = createUpdates({ app, publish, log, testing, notificationUrl: defaults.url, prepareInstall: async () => {
      saveWindow();
      siteSession.flushStorageData();
      await siteSession.cookies.flushStore();
      // Stop an owned server before NSIS starts waiting for our processes.
      if (localServer?.owned) {
        await localServer.stop();
        localServer = undefined;
      }
    } });
    ipcMain.handle('desktop:update', (event, action) => { requireShell(event); return updates.action(action); });
    ipcMain.handle('desktop:state', (event) => { requireShell(event); return state; });
    ipcMain.handle('desktop:control', (event, action) => {
      requireShell(event);
      if (action === 'minimize') win.minimize();
      else if (action === 'maximize') {
        toggleMaximize();
      }
      else if (action === 'close') win.close();
      else if (action === 'reload') void connect(state.url);
      else if (action === 'logs') void shell.openPath(path.join(app.getPath('userData'), 'desktop.log'));
      else throw new Error('Неизвестная команда окна');
    });
    createWindow();
    updates.start();
    app.once('will-quit', () => { updates.dispose(); downloads.dispose(); });
    app.on('activate', () => { if (!win) createWindow(); });
  }).catch((error) => { log(error.stack); app.quit(); });
  app.on('window-all-closed', () => { if (process.platform !== 'darwin') app.quit(); });
  app.on('before-quit', (event) => {
    if (shuttingDown || !localServer?.owned) return;
    event.preventDefault(); shuttingDown = true;
    void localServer.stop().finally(() => app.quit());
  });
}

const INSET = 10;
const TITLEBAR = 42;
const RADIUS = 0;

function siteBounds(width, height, expanded) {
  const inset = expanded ? 0 : INSET;
  return { x: inset, y: TITLEBAR, width: Math.max(1, width - inset * 2), height: Math.max(1, height - TITLEBAR - inset) };
}

function makeWindowRectangular(window) {
  // Native resizable windows need their system hit-test region outside the
  // client area. Only fixed-size dialogs need an explicit rectangular region.
  if (process.platform !== 'win32' || window.isResizable()) return;
  let previousWidth, previousHeight;
  const sync = () => {
    if (window.isDestroyed()) return;
    const [width, height] = window.getSize();
    if (width === previousWidth && height === previousHeight) return;
    previousWidth = width; previousHeight = height;
    // Windows 10 can clip the top corners of a caption-style window even
    // when roundedCorners is false. An explicit region fills the whole frame.
    window.setShape([{ x: 0, y: 0, width, height }]);
  };
  window.on('resize', sync);
  sync();
}

module.exports = { RADIUS, siteBounds, makeWindowRectangular };

const api = window.maikoDesktop;
const element = (id) => document.getElementById(id);

function render(state) {
  document.body.dataset.phase = state.phase;
  document.body.classList.toggle('maximized', state.maximized);
  element('status').textContent = state.phase === 'ready' ? 'Подключено' : state.phase === 'error' ? 'Нет подключения' : 'Подключаемся…';
  element('message').textContent = state.message;
  element('heading').textContent = state.phase === 'error' ? 'Не удалось подключиться' : 'Почти на месте';
  element('loader').hidden = state.phase !== 'loading';
  element('error-actions').hidden = state.phase !== 'error';
  element('connection-screen').hidden = state.phase === 'ready';
  element('maximize').title = state.maximized ? 'Восстановить' : 'Развернуть';
  element('maximize').setAttribute('aria-label', element('maximize').title);
  const files = state.downloads?.entries || [];
  const pending = files.filter((file) => file.phase === 'downloading');
  element('downloads-count').hidden = pending.length === 0;
  element('downloads-count').textContent = pending.length;
  element('downloads-action').title = pending.length ? `Открыть папку загрузок. Скачиваем: ${pending.length}` : 'Открыть папку загрузок';
  element('downloads-status').textContent = files[0] ? `${files[0].name}: ${files[0].message}` : '';
  const update = state.update;
  if (update) {
    element('app-version').textContent = `v${update.currentVersion}`;
    element('app-version').title = `Версия приложения ${update.currentVersion}`;
    element('update-status').textContent = update.message;
    element('update-dock').dataset.phase = update.phase;
    element('update-dock').title = update.message;
    const busy = ['checking', 'downloading', 'downloaded', 'installing'].includes(update.phase);
    const command = update.phase === 'error' && update.retryAction === 'apply' ? 'apply' :
      update.phase === 'available' || (update.phase === 'error' && update.retryAction === 'download') ? 'download' : 'check';
    element('update-action').dataset.command = command;
    element('update-action').disabled = busy || update.phase === 'disabled';
    const label = busy ? update.phase === 'checking' ? 'Проверка…' : ['downloaded', 'installing'].includes(update.phase) ? 'Установка…' : `${update.progress}%` :
      update.phase === 'error' ? 'Повторить' : command === 'download' ? 'Обновить' : update.phase === 'current' ? 'Обновлено' : 'Проверить';
    const description = command === 'apply' ? 'Повторить установку обновления' :
      command === 'download' ? `Обновить до версии ${update.version} с автоматическим перезапуском` : 'Проверить обновления';
    element('update-label').textContent = label;
    element('update-action').setAttribute('aria-label', busy ? update.message : description);
    element('update-action').title = update.phase === 'disabled' ? update.message : `${description}. ${update.message}`;
    element('update-progress').hidden = update.phase !== 'downloading';
    element('update-progress').value = update.progress;
  }
}

const { contextBridge, ipcRenderer } = require('electron');

// Only local shell/picker pages have this preload. The site has no Node bridge.
contextBridge.exposeInMainWorld('maikoDesktop', Object.freeze({
  state: () => ipcRenderer.invoke('desktop:state'),
  control: (action) => ipcRenderer.invoke('desktop:control', action),
  update: (action) => ipcRenderer.invoke('desktop:update', action),
  downloads: (command, id) => ipcRenderer.invoke('desktop:downloads', command, id),
  downloadsPanel: (open, anchor) => ipcRenderer.invoke('desktop:downloads-panel', open, anchor),
  onDownloadsPanel: (callback) => {
    const handler = (_event, open) => callback(open);
    ipcRenderer.on('desktop:downloads-panel-visible', handler);
    return () => ipcRenderer.removeListener('desktop:downloads-panel-visible', handler);
  },
  captureSources: () => ipcRenderer.invoke('desktop:capture-sources'),
  captureSelect: (choice) => ipcRenderer.invoke('desktop:capture-select', choice),
  onState: (callback) => {
    const handler = (_event, state) => callback(state);
    ipcRenderer.on('desktop:state-changed', handler);
    return () => ipcRenderer.removeListener('desktop:state-changed', handler);
  },
}));

const { BrowserWindow, desktopCapturer, ipcMain } = require('electron');
const path = require('node:path');
const { pathToFileURL } = require('node:url');
const { RADIUS, makeWindowRectangular } = require('./window-geometry.cjs');
const { windowProcesses, audioPolicy } = require('./capture-audio.cjs');

function configureCapture({ siteSession, getWindow, getSite, trusted, testing, log }) {
  let pending;
  let choosing = false, completedAudioPolicy;
  const pickerUrl = pathToFileURL(path.join(__dirname, 'ui', 'capture.html')).href;
  function requirePicker(event) {
    if (!pending || event.sender !== pending.window.webContents || event.senderFrame !== event.sender.mainFrame || event.senderFrame.url !== pickerUrl) {
      throw new Error('Нет активного выбора экрана');
    }
  }
  ipcMain.handle('desktop:capture-sources', (event) => {
    requirePicker(event);
    return { sources: pending.sources.map((source) => ({
      id: source.id, name: source.name, thumbnail: source.thumbnail.toDataURL(),
      audioKind: pending.audioRequested ? pending.policies.get(source.id)?.kind || null : null,
    })), audioAvailable: pending.audioRequested };
  });
  ipcMain.handle('desktop:capture-select', async (event, choice) => {
    requirePicker(event);
    const selection = pending;
    if (selection.selecting && choice !== null) return;
    const source = pending.sources.find((entry) => entry.id === choice?.id);
    if (choice !== null && !source) throw new Error('Источник больше не доступен');
    if (!source) { pending.finish(null); return; }
    selection.selecting = true;
    // Refresh the owning process at selection time: a window may have closed
    // or its HWND may have been reused while the chooser was open.
    let policy = null;
    if (choice.audio === true && pending.audioRequested) {
      try { policy = audioPolicy(source, await windowProcesses([source])); }
      catch (error) { log(`Screen audio unavailable: ${error.message}`); }
    }
    if (pending !== selection) return;
    completedAudioPolicy = policy ? { isolated: true, kind: policy.kind } : null;
    pending.finish({ video: source, ...(policy ? { audio: policy.device } : {}) });
  });
  ipcMain.handle('desktop:capture-audio-policy', (event) => {
    if (event.sender !== getSite()?.webContents || event.senderFrame !== event.sender.mainFrame || !trusted(event.senderFrame.url)) {
      throw new Error('Недоступно для этой страницы');
    }
    const policy = completedAudioPolicy;
    completedAudioPolicy = undefined;
    return policy || null;
  });
  siteSession.setDisplayMediaRequestHandler(async (request, callback) => {
    if (choosing || !request.frame || request.frame !== getSite()?.webContents.mainFrame || !trusted(request.securityOrigin)) {
      callback(null); return;
    }
    choosing = true;
    completedAudioPolicy = undefined;
    try {
      const sources = await desktopCapturer.getSources({ types: ['screen', 'window'], thumbnailSize: { width: 320, height: 180 }, fetchWindowIcons: false });
      if (!sources.length || !request.frame || request.frame !== getSite()?.webContents.mainFrame) { choosing = false; callback(null); return; }
      let owners = {};
      if (request.audioRequested) {
        try { owners = await windowProcesses(sources); }
        catch (error) { log(`Window audio lookup: ${error.message}`); }
      }
      const policies = new Map(sources.map((source) => [source.id, audioPolicy(source, owners)]));
      const picker = new BrowserWindow({
        parent: getWindow(), modal: true, width: 780, height: 600,
        frame: false, roundedCorners: false, transparent: false, thickFrame: false, backgroundColor: '#141416',
        resizable: false, minimizable: false, maximizable: false, show: false,
        title: 'Демонстрация экрана — Майко',
        webPreferences: { preload: path.join(__dirname, 'preload.cjs'), sandbox: true, contextIsolation: true, nodeIntegration: false },
      });
      makeWindowRectangular(picker);
      let finished = false;
      const finish = (streams) => {
        if (finished) return;
        finished = true;
        choosing = false;
        pending = undefined;
        if (!streams) completedAudioPolicy = undefined;
        if (!picker.isDestroyed()) picker.close();
        try { callback(request.frame && !getSite()?.webContents.isDestroyed() ? streams : null); }
        catch (error) { log(error.message); }
      };
      pending = { window: picker, sources, policies, finish, audioRequested: request.audioRequested && process.platform === 'win32' };
      picker.on('closed', () => finish(null));
      picker.webContents.setWindowOpenHandler(() => ({ action: 'deny' }));
      picker.webContents.on('will-navigate', (event, url) => { if (url !== pickerUrl) event.preventDefault(); });
      picker.contentView.setBackgroundColor('#141416');
      picker.contentView.setBorderRadius(RADIUS);
      picker.once('ready-to-show', () => { if (!testing) picker.show(); });
      await picker.loadURL(pickerUrl);
    } catch (error) {
      log(error.message);
      choosing = false;
      if (pending) pending.finish(null); else callback(null);
    }
  });
}

module.exports = { configureCapture };
